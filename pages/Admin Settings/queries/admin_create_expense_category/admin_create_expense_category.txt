WITH current_operator AS (
    SELECT require_app_user_id_by_email({{ appsmith.user.email }}) AS user_id
)
INSERT INTO expense_categories (parent_id, code, name, description, is_active)
SELECT
    NULLIF({{ SelectParentExpenseCategory.selectedOptionValue }}, '')::uuid,
    {{ InputExpenseCategoryCode.text }},
    {{ InputExpenseCategoryName.text }},
    NULLIF({{ TextAreaExpenseCategoryDescrip.text }}, ''),
    COALESCE({{ SwitchExpenseCategoryActive.isSwitchedOn }}, true)
FROM current_operator
RETURNING *;
