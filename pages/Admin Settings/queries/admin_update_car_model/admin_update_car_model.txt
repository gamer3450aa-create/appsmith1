WITH current_operator AS (
    SELECT require_app_user_id_by_email({{ appsmith.user.email }}) AS user_id
)
UPDATE car_models cm
SET car_brand_id = {{ SelectCarBrand.selectedOptionValue }}::uuid,
    name = {{ InputCarModelName.text }},
    start_year = NULLIF({{ InputStartYear.text }}, '')::integer,
    end_year = NULLIF({{ InputEndYear.text }}, '')::integer,
    is_active = COALESCE({{ SwitchCarModelActive.isSwitchedOn }}, true)
FROM current_operator
WHERE cm.id = {{ TableCarModels.selectedRow.id }}::uuid
RETURNING cm.*;
