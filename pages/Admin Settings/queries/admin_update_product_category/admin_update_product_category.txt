WITH current_operator AS (
    SELECT require_app_user_id_by_email({{ appsmith.user.email }}) AS user_id
)
UPDATE categories c
SET parent_id = NULLIF({{ SelectParentCategory.selectedOptionValue }}, '')::uuid,
    code = {{ InputCategoryCode.text }},
    name = {{ InputCategoryName.text }},
    description = NULLIF({{ TextAreaCategoryDescription.text }}, ''),
    is_active = COALESCE({{ SwitchCategoryActive.isSwitchedOn }}, true)
FROM current_operator
WHERE c.id = {{ TableProductCategories.selectedRow.id }}::uuid
RETURNING c.*;
