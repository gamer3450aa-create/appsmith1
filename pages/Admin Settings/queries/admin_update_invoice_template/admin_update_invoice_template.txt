WITH current_operator AS (
    SELECT require_app_user_id_by_email({{ appsmith.user.email }}) AS user_id
)
UPDATE invoice_templates it
SET branch_id = NULLIF({{ SelectBranch.selectedOptionValue }}, '')::uuid,
    template_name = {{ InputTemplateName.text }},
    invoice_type = {{ SelectInvoiceType.selectedOptionValue }},
    paper_size = COALESCE(NULLIF({{ SelectPaperSize.selectedOptionValue }}, ''), 'a4'),
    language = COALESCE(NULLIF({{ SelectLanguage.selectedOptionValue }}, ''), 'ar'),
    html_template = {{ TextAreaHtmlTemplate.text }},
    css_template = NULLIF({{ TextAreaCssTemplate.text }}, ''),
    show_logo = COALESCE({{ SwitchShowLogo.isSwitchedOn }}, true),
    show_tax_number = COALESCE({{ SwitchShowTaxNumber.isSwitchedOn }}, true),
    show_customer_info = COALESCE({{ SwitchShowCustomerInfo.isSwitchedOn }}, true),
    show_product_sku = COALESCE({{ SwitchShowProductSku.isSwitchedOn }}, true),
    show_barcode = COALESCE({{ SwitchShowBarcode.isSwitchedOn }}, false),
    show_qr_code = COALESCE({{ SwitchShowQrCode.isSwitchedOn }}, false),
    show_discount = COALESCE({{ SwitchShowDiscount.isSwitchedOn }}, true),
    show_tax = COALESCE({{ SwitchShowTax.isSwitchedOn }}, true),
    show_footer = COALESCE({{ SwitchShowFooter.isSwitchedOn }}, true),
    is_default = COALESCE({{ SwitchDefaultTemplate.isSwitchedOn }}, false),
    is_active = COALESCE({{ SwitchTemplateActive.isSwitchedOn }}, true)
FROM current_operator
WHERE it.id = {{ TableInvoiceTemplates.selectedRow.id }}::uuid
RETURNING it.*;
