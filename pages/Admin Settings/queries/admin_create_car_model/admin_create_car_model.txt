WITH current_operator AS (
    SELECT require_app_user_id_by_email({{ appsmith.user.email }}) AS user_id
)
INSERT INTO car_models (car_brand_id, name, start_year, end_year, is_active)
SELECT
    {{ SelectCarBrand.selectedOptionValue }}::uuid,
    {{ InputCarModelName.text }},
    NULLIF({{ InputStartYear.text }}, '')::integer,
    NULLIF({{ InputEndYear.text }}, '')::integer,
    COALESCE({{ SwitchCarModelActive.isSwitchedOn }}, true)
FROM current_operator
RETURNING *;
