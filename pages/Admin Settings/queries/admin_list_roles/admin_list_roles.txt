SELECT
    r.*,
    COALESCE(user_counts.active_user_count, 0)::INTEGER AS active_user_count
FROM roles r
LEFT JOIN (
    SELECT role_id, count(*)::INTEGER AS active_user_count
    FROM users
    WHERE is_active = true
    GROUP BY role_id
) user_counts ON user_counts.role_id = r.id
ORDER BY r.is_system DESC, r.name;
