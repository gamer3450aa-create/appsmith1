WITH current_operator AS (
    SELECT require_app_user_id_by_email({{ appsmith.user.email }}) AS user_id
)
UPDATE warehouses w
SET branch_id = {{ SelectBranch.selectedOptionValue }}::uuid,
    code = {{ InputWarehouseCode.text }},
    name = {{ InputWarehouseName.text }},
    address = NULLIF({{ TextAreaWarehouseAddress.text }}, ''),
    is_active = COALESCE({{ SwitchWarehouseActive.isSwitchedOn }}, true)
FROM current_operator
WHERE w.id = {{ TableWarehouses.selectedRow.id }}::uuid
RETURNING w.*;
