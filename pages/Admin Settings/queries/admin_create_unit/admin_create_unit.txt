WITH current_operator AS (
    SELECT require_app_user_id_by_email({{ appsmith.user.email }}) AS user_id
)
INSERT INTO units (code, name, decimal_places, is_active)
SELECT
    {{ InputUnitCode.text }},
    {{ InputUnitName.text }},
    COALESCE(NULLIF({{ InputDecimalPlaces.text }}, '')::smallint, 0),
    COALESCE({{ SwitchUnitActive.isSwitchedOn }}, true)
FROM current_operator
RETURNING *;
