WITH current_operator AS (
    SELECT require_app_user_id_by_email({{ appsmith.user.email }}) AS user_id
)
UPDATE expense_categories ec
SET parent_id = NULLIF({{ SelectParentExpenseCategory.selectedOptionValue }}, '')::uuid,
    code = {{ InputExpenseCategoryCode.text }},
    name = {{ InputExpenseCategoryName.text }},
    description = NULLIF({{ TextAreaExpenseCategoryDescrip.text }}, ''),
    is_active = COALESCE({{ SwitchExpenseCategoryActive.isSwitchedOn }}, true)
FROM current_operator
WHERE ec.id = {{ TableExpenseCategories.selectedRow.id }}::uuid
RETURNING ec.*;
