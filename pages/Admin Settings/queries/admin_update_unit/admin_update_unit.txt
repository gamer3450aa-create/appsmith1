WITH current_operator AS (
    SELECT require_app_user_id_by_email({{ appsmith.user.email }}) AS user_id
)
UPDATE units u
SET code = {{ InputUnitCode.text }},
    name = {{ InputUnitName.text }},
    decimal_places = COALESCE(NULLIF({{ InputDecimalPlaces.text }}, '')::smallint, 0),
    is_active = COALESCE({{ SwitchUnitActive.isSwitchedOn }}, true)
FROM current_operator
WHERE u.id = {{ TableUnits.selectedRow.id }}::uuid
RETURNING u.*;
