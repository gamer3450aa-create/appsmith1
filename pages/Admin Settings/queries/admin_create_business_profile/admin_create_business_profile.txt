WITH current_operator AS (
    SELECT require_app_user_id_by_email({{ appsmith.user.email }}) AS user_id
)
INSERT INTO business_profiles (
    branch_id,
    business_name,
    legal_name,
    tax_number,
    commercial_register,
    phone,
    whatsapp,
    email,
    website,
    address,
    city,
    country,
    logo_url,
    default_currency,
    invoice_footer,
    return_policy,
    is_default,
    is_active
)
SELECT
    NULLIF({{ SelectBranch.selectedOptionValue }}, '')::uuid,
    {{ InputBusinessName.text }},
    NULLIF({{ InputLegalName.text }}, ''),
    NULLIF({{ InputTaxNumber.text }}, ''),
    NULLIF({{ InputCommercialRegister.text }}, ''),
    NULLIF({{ InputPhone.text }}, ''),
    NULLIF({{ InputWhatsapp.text }}, ''),
    NULLIF({{ InputEmail.text }}, ''),
    NULLIF({{ InputAreaAddress.text }}, ''),
    NULLIF({{ InputAreaAddress.text }}, ''),
    NULLIF({{ InputCity.text }}, ''),
    COALESCE(NULLIF({{ InputCountry.text }}, ''), 'Egypt'),
    NULLIF({{ InputLogoUrl.text }}, ''),
    COALESCE(NULLIF({{ TextAreaReturnPolicy.text }}, ''), 'EGP'),
    NULLIF({{ TextAreaInvoiceFooter.text }}, ''),
    NULLIF({{ TextAreaReturnPolicy.text }}, ''),
    COALESCE({{ SwitchDefaultProfile.isSwitchedOn }}, false),
    COALESCE({{ SwitchActiveProfile.isSwitchedOn }}, true)
FROM current_operator
RETURNING *;
