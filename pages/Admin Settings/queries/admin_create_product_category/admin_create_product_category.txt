WITH current_operator AS (
    SELECT require_app_user_id_by_email({{ appsmith.user.email }}) AS user_id
)
INSERT INTO categories (parent_id, code, name, description, is_active)
SELECT
    NULLIF({{ SelectParentCategory.selectedOptionValue }}, '')::uuid,
    {{ InputCategoryCode.text }},
    {{ InputCategoryName.text }},
    NULLIF({{ TextAreaCategoryDescription.text }}, ''),
    COALESCE({{ SwitchCategoryActive.isSwitchedOn }}, true)
FROM current_operator
RETURNING *;
