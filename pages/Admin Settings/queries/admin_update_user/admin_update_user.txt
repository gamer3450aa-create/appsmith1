WITH current_operator AS (
    SELECT require_app_user_id_by_email({{ appsmith.user.email }}) AS user_id
)
UPDATE users u
SET role_id = {{ SelectRole.selectedOptionValue }}::uuid,
    branch_id = NULLIF({{ SelectBranch.selectedOptionValue }}, '')::uuid,
    full_name = {{ InputUserFullName.text }},
    email = lower(trim({{ InputUserEmail.text }})),
    phone = NULLIF({{ InputUserPhone.text }}, ''),
    is_active = COALESCE({{ SwitchUserActive.isSwitchedOn }}, true)
FROM current_operator
WHERE u.id = {{ TableUsers.selectedRow.id }}::uuid
RETURNING u.id, u.role_id, u.branch_id, u.full_name, u.email, u.phone, u.is_active, u.created_at, u.updated_at;
