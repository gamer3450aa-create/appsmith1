SELECT *
FROM vw_finance_customer_open_invoices
WHERE (
      NULLIF({{ SelectBranch.selectedOptionValue }}, '') IS NULL
      OR branch_id = NULLIF({{ SelectBranch.selectedOptionValue }}, '')::uuid
  )
  AND (
      NULLIF({{ SelectCustomer.selectedOptionValue }}, '') IS NULL
      OR customer_id = NULLIF({{ SelectCustomer.selectedOptionValue }}, '')::uuid
  )
ORDER BY invoice_date DESC, invoice_number DESC
LIMIT 200;
