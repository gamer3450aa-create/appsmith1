SELECT *
FROM vw_finance_supplier_open_invoices
WHERE (
      NULLIF({{ SelectBranch.selectedOptionValue }}, '') IS NULL
      OR branch_id = NULLIF({{ SelectBranch.selectedOptionValue }}, '')::uuid
  )
  AND (
      NULLIF({{ SelectSupplier.selectedOptionValue }}, '') IS NULL
      OR supplier_id = NULLIF({{ SelectSupplier.selectedOptionValue }}, '')::uuid
  )
ORDER BY invoice_date DESC, invoice_number DESC
LIMIT 200;
