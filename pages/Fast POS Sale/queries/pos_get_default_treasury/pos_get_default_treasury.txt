WITH cfg AS (
    SELECT
        (SELECT NULLIF(setting_value, '')::uuid FROM system_settings WHERE setting_key = 'default_branch_id') AS default_branch_id,
        (SELECT NULLIF(setting_value, '')::uuid FROM system_settings WHERE setting_key = 'default_treasury_id') AS default_treasury_id
),
f AS (
    SELECT COALESCE(
        NULLIF({{ SelectBranch.selectedOptionValue }}, '')::uuid,
        (SELECT default_branch_id FROM cfg)
    ) AS branch_id
)
SELECT
    t.id AS value,
    t.name AS label
FROM treasuries t
CROSS JOIN cfg
CROSS JOIN f
WHERE t.is_active = true
  AND t.branch_id = f.branch_id
ORDER BY
    CASE WHEN t.id = cfg.default_treasury_id THEN 0 ELSE 1 END,
    t.created_at ASC
LIMIT 1;
