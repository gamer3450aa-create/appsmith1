INSERT INTO sales_invoice_items (
    sales_invoice_id,
    product_id,
    line_number,
    raw_item_name,
    quantity,
    unit_price,
    unit_cost_at_sale,
    discount_amount,
    tax_amount,
    line_total,
    profit_amount,
    notes
)
SELECT
    si.id,
    {{ TableProductResults.selectedRow.product_id }}::uuid,
    COALESCE(
        (
            SELECT max(line_number) + 1
            FROM sales_invoice_items
            WHERE sales_invoice_id = si.id
        ),
        1
    ),
    {{ TableProductResults.selectedRow.product_name }},
    COALESCE(NULLIF({{ InputCartQuantity.text }}, '')::numeric, 1),
    {{ TableProductResults.selectedRow.sale_price }}::numeric,
    0,
    0,
    0,
    round(
        COALESCE(NULLIF({{ InputCartQuantity.text }}, '')::numeric, 1)
        * {{ TableProductResults.selectedRow.sale_price }}::numeric,
        2
    ),
    round(
        COALESCE(NULLIF({{ InputCartQuantity.text }}, '')::numeric, 1)
        * {{ TableProductResults.selectedRow.sale_price }}::numeric,
        2
    ),
    NULL
FROM sales_invoices si
WHERE si.id = {{ appsmith.store.currentSalesInvoiceId }}::uuid
  AND si.status IN ('draft', 'pending_approval')
RETURNING *;
