SELECT *
FROM vw_expenses_summary
WHERE status = 'approved'
  AND expense_date = COALESCE(NULLIF({{ DatePickerShiftDate.selectedDate }}, '')::date, CURRENT_DATE)
  AND (
      NULLIF({{ SelectBranch.selectedOptionValue }}, '') IS NULL
      OR branch_id = NULLIF({{ SelectBranch.selectedOptionValue }}, '')::uuid
  )
ORDER BY expense_date DESC, expense_number DESC;
