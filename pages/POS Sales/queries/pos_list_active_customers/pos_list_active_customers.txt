SELECT id AS value, code || ' - ' || name AS label
FROM customers
WHERE is_active = true
  AND (
      NULLIF({{ SelectCustomer.searchText }}, '') IS NULL
      OR code ILIKE '%' || {{ SelectCustomer.searchText }} || '%'
      OR name ILIKE '%' || {{ SelectCustomer.searchText }} || '%'
      OR phone ILIKE '%' || {{ SelectCustomer.searchText }} || '%'
  )
ORDER BY name
LIMIT 50;
