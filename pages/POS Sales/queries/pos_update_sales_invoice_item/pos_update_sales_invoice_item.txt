UPDATE sales_invoice_items sii
SET quantity = {{ InputCartQuantity.text }}::numeric,
    unit_price = {{ InputUnitPrice.text }}::numeric,
    discount_amount = COALESCE(NULLIF({{ InputLineDiscount.text }}, '')::numeric, 0),
    tax_amount = COALESCE(NULLIF({{ InputLineTax.text }}, '')::numeric, 0),
    line_total = round(
        ({{ InputCartQuantity.text }}::numeric * {{ InputUnitPrice.text }}::numeric)
        - COALESCE(NULLIF({{ InputLineDiscount.text }}, '')::numeric, 0)
        + COALESCE(NULLIF({{ InputLineTax.text }}, '')::numeric, 0),
        2
    ),
    profit_amount = round(
        ({{ InputCartQuantity.text }}::numeric * {{ InputUnitPrice.text }}::numeric)
        - COALESCE(NULLIF({{ InputLineDiscount.text }}, '')::numeric, 0)
        + COALESCE(NULLIF({{ InputLineTax.text }}, '')::numeric, 0),
        2
    ),
    notes = NULLIF({{ TextAreaLineNotes.text }}, '')
FROM sales_invoices si
WHERE sii.id = {{ TableCart.selectedRow.sales_invoice_item_id }}::uuid
  AND si.id = sii.sales_invoice_id
  AND si.status IN ('draft', 'pending_approval')
RETURNING sii.*;
