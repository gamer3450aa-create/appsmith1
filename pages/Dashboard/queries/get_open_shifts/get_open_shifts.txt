SELECT
    shift_id,
    shift_number,
    branch_id,
    branch_name,
    treasury_id,
    treasury_name,
    opened_by_name,
    opened_at,
    opening_cash,
    expected_cash
FROM vw_shifts_summary
WHERE status = 'open'
  AND (
      NULLIF({{ SelectBranch.selectedOptionValue }}, '') IS NULL
      OR branch_id = NULLIF({{ SelectBranch.selectedOptionValue }}, '')::uuid
  )
ORDER BY opened_at DESC;
